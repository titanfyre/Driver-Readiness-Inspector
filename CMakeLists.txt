cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(DriverReadinessInspector VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 with the Visual Studio C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ENABLE_ASAN "Enable AddressSanitizer for offline validation" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc)
if(ENABLE_ASAN)
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/INCREMENTAL:NO /DEBUG:FULL)
endif()
add_executable(DriverReadinessInspector WIN32 App/Main.cpp App/Model.cpp App/Collect.cpp Tests/Fixtures.cpp)
target_link_libraries(DriverReadinessInspector PRIVATE setupapi cfgmgr32 bcrypt crypt32 wintrust wevtapi advapi32 user32 shell32 comctl32 comdlg32 gdi32 dwmapi uxtheme ole32)
target_link_options(DriverReadinessInspector PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'" "/PDBALTPATH:%_PDB%")
set_target_properties(DriverReadinessInspector PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_test(NAME OfflineEscapingSchemaStatusFixtures COMMAND DriverReadinessInspector --self-test)
add_test(NAME OfflineJsonReport COMMAND DriverReadinessInspector --preview --report "${CMAKE_BINARY_DIR}/preview-report-$<CONFIG>.json")
add_test(NAME OfflineHtmlReport COMMAND DriverReadinessInspector --preview --report "${CMAKE_BINARY_DIR}/preview-report-$<CONFIG>.html" --format=html)
